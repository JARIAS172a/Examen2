-- Tablas propias del sistema (además de Identity)
-- Productos
CREATE TABLE Productos (
    Id INT IDENTITY PRIMARY KEY,
	IdUsuarioCreador NVARCHAR(MAX)		NOT NULL,
    Nombre NVARCHAR(100)				NOT NULL,
    Descripcion NVARCHAR(255)			NOT NULL,
    Precio DECIMAL(10,2)				NOT NULL,
	CONSTRAINT FK_Productos_Usuarios FOREIGN KEY (IdUsuarioCreador) REFERENCES AspNetusers(Id)
);

-- Maquinaria
CREATE TABLE Maquinarias (
    Id INT IDENTITY PRIMARY KEY,	
	IdUsuarioCreadorM NVARCHAR(MAX)		NOT NULL,
    Nombre NVARCHAR(100),
    Descripcion NVARCHAR(255),
    Marca NVARCHAR(50),
    Modelo NVARCHAR(50),
	CONSTRAINT FK_Maquinarias_Usuarios FOREIGN KEY (IdUsuarioCreadorM) REFERENCES AspNetusers(Id)
);

-- Preguntas
CREATE TABLE Preguntas (
    Id INT IDENTITY PRIMARY KEY,	
	IdUsuarioCreadorP NVARCHAR(MAX)		NOT NULL,
    Titulo NVARCHAR(100),
    Contenido NVARCHAR(MAX),
    Fecha DATETIME DEFAULT GETDATE(),
	CONSTRAINT FK_Preguntas_Usuarios FOREIGN KEY (IdUsuarioCreadorP) REFERENCES AspNetusers(Id)
);
